Wartość -1 sygnalizuje "brak danych" i nie koliduje z prawdziwą skalą 0–8.

vei: po fillna(-1) nie ma już NaN → cast do int64 metodą .astype(int)
deaths_total: po fillna(0) → cast do int64
year, elevation: już int64 — nic nie rób
month, day: zostaw jako float64 z NaN — nie castuj do int (int nie przyjmuje NaN w standardowym pandas)

def transform_noaa() -> pd.DataFrame:
    """Clean NOAA volcanic eruptions data. Returns Silver-layer DataFrame."""
    df = pd.read_json(RAW_DIR / NOAA_FILENAME)
    logging.info(f"NOAA raw: {df.shape}")

    columns = ['year', 'month', 'day', 'name', 'country', 'location',
               'latitude', 'longitude', 'elevation', 'morphology',
               'vei', 'deathsTotal', 'significant']
    df = df[columns]

    df = df.rename(columns={'deathsTotal': 'deaths_total'})

    df['deaths_total'] = df['deaths_total'].fillna(0)
    df['vei'] = df['vei'].fillna(-1)

    df['vei'] = df['vei'].astype(int)
    df['deaths_total'] = df['deaths_total'].astype(int)

    before = len(df)
    df = df.drop_duplicates()
    logging.info(f"Removed {before - len(df)} duplicates")

    df = df.reset_index(drop=True)
    logging.info(f"NOAA clean: {df.shape}")
    return df
